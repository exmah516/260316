def recv_exact(sock: socket.socket, size: int) -> bytes:
    data = bytearray()
    while len(data) < size:

        if not chunk:
            raise ConnectionError("socket closed by peer")
        data.extend(chunk)

def parse_channels(resp: bytes) -> Optional[List[float]]:
    if len(resp) != RESP_LEN:
        return None
    vals: List[float] = []
    for i in range(CHANNEL_COUNT):

        lo = resp[CHANNEL_OFFSET + i * 2 + 1]
        raw = int.from_bytes(bytes([hi, lo]), byteorder="big", signed=True)
        vals.append(raw / 1000.0)
    return vals


def open_socket(ip: str, port: int, timeout_s: float) -> socket.socket:

def collect_samples(ip: str, port: int, seconds: float, timeout_ms: int) -> Tuple[List[Tuple], Dict[str, int]]:
    timeout_s = max(timeout_ms, 1) / 1000.0
    end_t = time.perf_counter() + max(seconds, 0.1)

        "io_error": 0,
        "bad_length": 0,

    try:
        while time.perf_counter() < end_t:

            "effective_hz": float("nan"),
            "inst_hz_mean": float("nan"),

def rank_likely_sensor_channels(channels: np.ndarray) -> Tuple[List[int], List[Dict[str, float]]]:
    if channels.size == 0:
        return [], []

            "p2p": p2p_v,
            "mean_abs_diff": mad_v,
            "score": 0.0,
        })

    std_arr = np.array([m["std"] for m in metrics], dtype=np.float64)
    p2p_arr = np.array([m["p2p"] for m in metrics], dtype=np.float64)
    mad_arr = np.array([m["mean_abs_diff"] for m in metrics], dtype=np.float64)

    def norm(a: np.ndarray) -> np.ndarray:
        m = np.max(a)
        if m <= 1e-12:
            return np.zeros_like(a)
        return a / m

    std_n = norm(std_arr)
    p2p_n = norm(p2p_arr)
    mad_n = norm(mad_arr)

    score = 0.4 * std_n + 0.3 * p2p_n + 0.3 * mad_n
    for i, s in enumerate(score):
        metrics[i]["score"] = float(s)

    order = np.argsort(score)[::-1]
    likely = [int(order[0])]
    if order.size > 1:
        likely.append(int(order[1]))

    metrics_sorted = [metrics[int(i)] for i in order]
    return likely, metrics_sorted
